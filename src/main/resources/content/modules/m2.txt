public class View extends JFrame {

    public void initialise() {
        setTitle("Übung");
        setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        addComponents();
        pack();
        setVisible(true);
    }

    private void addComponents() {
        JPanel keys = new JPanel(new GridLayout(4, 3, 5, 5));
        String[] labels = {"7", "8", "9", "4", "5", "6", "1", "2", "3", "0", ".", "C"};
        for (String label : labels) {
            keys.add(new JButton(label));
        }
        add(keys, BorderLayout.CENTER);
    }

    public static void main(String[] args) {
        new View().initialise();
    }
}

        addComponents();

public class View extends JFrame {

    public void initialise() {
        setTitle("Übung");
        setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        addComponents();
        pack();
        setVisible(true);
    }

    private void addComponents() {
        JPanel top = new JPanel(new FlowLayout(FlowLayout.CENTER, 25, 10));
        JPanel center = new JPanel(new GridLayout(4, 3, 5, 5));
        center.setBorder(BorderFactory.createEmptyBorder(0, 20, 0, 0));
        JPanel right = new JPanel(new GridLayout(4, 1));
        add(top, BorderLayout.NORTH);
        add(center, BorderLayout.CENTER);
        add(right, BorderLayout.EAST);
        // Damit man etwas sieht, darfst du in die Panels Buttons legen.
    }

    public static void main(String[] args) {
        new View().initialise();
    }
