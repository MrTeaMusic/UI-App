@@solution Controller.java
import javax.swing.*;

public class Controller {
    private Model model;
    private View view;

    private void readGrades() {
        JTextField[] fields = view.getFields();
        int[] grades = new int[fields.length];
        try {
            for (int i = 0; i < fields.length; i++) {
                grades[i] = Integer.parseInt(fields[i].getText().trim());
            }
        } catch (NumberFormatException e) {
            view.showError();
            return;
        }
        model.setGrades(grades);
    }
}

public class Main {

    public static double parseDecimal(String text) {
        try {
            return Double.parseDouble(text.replace(',', '.'));
        } catch (NumberFormatException e) {
            return 0.0;
        }
    }

    public static void main(String[] args) {
        System.out.println(parseDecimal("3,5"));  // 3.5
        System.out.println(parseDecimal("2.25")); // 2.25
        System.out.println(parseDecimal("abc"));  // 0.0
    }
}

public class Main {

    public static void main(String[] args) {
        Locale locale = args.length > 0 ? new Locale(args[0]) : Locale.ENGLISH;
        ResourceBundle bundle = null;
        try {
            bundle = ResourceBundle.getBundle("Messages", locale);
        } catch (MissingResourceException e) {
            System.err.println("Sprachdatei nicht gefunden");
            System.exit(1);
        }
        System.out.println(bundle.getString("title"));
    }
}
