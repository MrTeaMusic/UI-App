public class Main {

    public static double average(int[] values) {
        int sum = 0;
        for (int v : values) {
            sum += v;
        }
        return sum / (double) values.length;
    }

    public static void main(String[] args) {
        System.out.println(average(new int[]{1, 2}));   // erwartet: 1.5
        System.out.println(average(new int[]{3, 4, 5})); // erwartet: 4.0
    }
}

public class Main {

    public static int parseOrZero(String text) {
        try {
            return Integer.parseInt(text.trim());
        } catch (NumberFormatException e) {
            return 0;
        }
    }

    public static void main(String[] args) {
        System.out.println(parseOrZero("42"));   // erwartet: 42
        System.out.println(parseOrZero(" 7 "));  // erwartet: 7
        System.out.println(parseOrZero("abc"));  // erwartet: 0
    }

public class Main {

    public static String label(String cmd) {
        switch (cmd) {
            case "S":
                return "Start";
            case "P":
                return "Pause";
            case "E":
                return "Ende";
            default:
                return "?";
        }
    }

    public static void main(String[] args) {
        System.out.println(label("S")); // Start
        System.out.println(label("P")); // Pause
        System.out.println(label("E")); // Ende
        System.out.println(label("X")); // ?
    }
