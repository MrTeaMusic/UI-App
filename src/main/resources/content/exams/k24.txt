import java.text.DecimalFormat;

public class Model {
	private double meter;
	private double inch;
	private double foot;
	private double yard;
	
	private String unit;
	private StringBuilder builder;

	public void initialise(double value, String unit) {
		this.unit = unit;
		this.builder = new StringBuilder();
		
		calculate(value);
	}
	
	public double getMeter() {
		return meter;
	}

	public void setMeter(double meter) {
		this.meter = meter;
		inch = 1. / 0.0254 * meter;
		foot = 1. / 0.3048 * meter;
		yard = 1. / 0.9144 * meter;
	}

	public double getInch() {
		return inch;
	}

	public void setInch(double inch) {
		this.inch = inch;
		
		meter = 0.0254 * inch;
		foot = 1. / 12. * inch;
		yard = 1. / 36. * inch;
	}

	public double getFoot() {
		return foot;
	}

	public void setFoot(double foot) {
		this.foot = foot;
		meter = 0.3048 * foot;
		inch = 12 * foot;
		yard = 1. / 3. * foot;
	}

	public double getYard() {
		return yard;
	}

	public void setYard(double yard) {
		this.yard = yard;
		meter = 0.9144 * yard;
		inch = 36 * yard;
		foot = 3 * yard;
	}

	public void setUnit(String unit) {
		double value = 0;
		
		switch (this.unit) {
			case "m":
				value = getMeter();
				break;
			case "inch":
				value = getInch();
				break;				
			case "foot":
				value = getFoot();
				break;
			case "yard":
				value = getYard();
				break;
		}
		
		this.unit = unit;
		calculate(value);
	}

	public StringBuilder getStringBuilder() {
		return builder;
	}

	public void calculate(double value) {
		switch (unit) {
			case "m":
				setMeter(value);
				break;
			case "inch":
				setInch(value);
				break;				
			case "foot":
				setFoot(value);
				break;
			case "yard":
				setYard(value);
				break;
		}
	}
	
	public static String format(double value) {
		return new DecimalFormat("#.####").format(value);
	}
}

        add(right, BorderLayout.EAST);

        unitBox.setActionCommand("choose");
        unitBox.addActionListener(controller);
        top.add(inputField);
        top.add(unitBox);

        String[] keys = {"7","8","9","4","5","6","1","2","3","0",".","C"};
        for (String key : keys) {
            JButton b = new JButton(key);
            b.addActionListener(controller);
            center.add(b);
        }
        JButton calc = new JButton(bundle.getString("calculate"));
        calc.setActionCommand("Calculate"); // Text ist übersetzt, das Kommando bleibt gleich
        calc.addActionListener(controller);
        bottom.add(calc);

        meterField = addField(right, "m");
        inchField  = addField(right, "inch");
        footField  = addField(right, "foot");
        yardField  = addField(right, "yard");
    }

    private JTextField addField(JPanel parent, String label) {
        JLabel lbl = new JLabel(label);
        JTextField field = new JTextField(20);
        field.setFont(new Font(Font.MONOSPACED, Font.PLAIN, 12));
        field.setHorizontalAlignment(JTextField.RIGHT);
        field.setEditable(false);
        JPanel row = new JPanel(new FlowLayout(FlowLayout.LEFT, 20, 5));
        row.add(field);
        row.add(lbl);
        parent.add(row);
        return field;

        super.paint(g);

        inchField.setText(Model.format(model.getInch()));
        footField.setText(Model.format(model.getFoot()));
        yardField.setText(Model.format(model.getYard()));
    }

    public JComboBox<String> getUnitBox() { return unitBox; }

@@solution Controller.java
import java.awt.event.*;
public class Controller implements ActionListener {
    private Model model;
    private View view;

    public void initialise(Model model, View view) {
        this.model = model;
        this.view = view;
    }

    private static boolean isNumberBetweenZeroAndNine(String str) {
        try {
            int i = Integer.valueOf(str);
            return i >= 0 && i <= 9;
        } catch (NumberFormatException e) {
            return false;
        }
    }

    @Override
    public void actionPerformed(ActionEvent e) {
        StringBuilder sb = model.getStringBuilder();
        sb.setLength(0);
        sb.append(view.getInputField().getText());

        String cmd = e.getActionCommand();
        if (isNumberBetweenZeroAndNine(cmd)) {
            sb.append(cmd);
            view.getInputField().setText(sb.toString());
        } else if (cmd.equals(".")) {
            if (sb.indexOf(".") < 0) {
                sb.append(cmd);
                view.getInputField().setText(sb.toString());
            }
        } else if (cmd.equals("C")) {
            reset();
        } else if (cmd.equals("Calculate")) {
            try {
                String text = sb.toString().replace(',', '.');
                model.calculate(Double.parseDouble(text));
            } catch (NumberFormatException ex) {
                reset();
            }
        } else if (cmd.equals("choose")) {
            model.setUnit((String) view.getUnitBox().getSelectedItem());
        }
        view.repaint();
    }

    private void reset() {
        model.initialise(0, (String) view.getUnitBox().getSelectedItem());
        model.getStringBuilder().setLength(0);
        view.getInputField().setText("");
    }
}

                model.calculate(Double.parseDouble(text));
            } catch (NumberFormatException ex) {
                reset();
            }
        } else if (cmd.equals("choose")) {
            model.setUnit((String) view.getUnitBox().getSelectedItem());
        }
        view.repaint();
    }

    private void reset() {
        model.initialise(0, (String) view.getUnitBox().getSelectedItem());
        model.getStringBuilder().setLength(0);
        view.getInputField().setText("");
    }
    // isNumberBetweenZeroAndNine: wie gegeben
}
