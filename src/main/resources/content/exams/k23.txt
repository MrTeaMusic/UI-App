public class JDiagram {
    private static int BASE_PADDING = 10;
    private static double PERCENTAGE_RADIUS = 0.5;

    private Point getPadding() {
        int width = getWidth();
        int height = getHeight();

        if (width > height) {
            return new Point((width - height) / 2 + JDiagram.BASE_PADDING, JDiagram.BASE_PADDING);
        } else if (width < height) {
            return new Point(JDiagram.BASE_PADDING, (height - width) / 2 + JDiagram.BASE_PADDING);
        }

        return new Point(JDiagram.BASE_PADDING, JDiagram.BASE_PADDING);
    }

    private Point getCenter() {
        return new Point(getX() + getWidth()/2, getY() + getHeight()/2);
    }

    private Point getCrossPoint(double percentage, double percentageRadius) {
        Point padding = getPadding();
        double r = percentageRadius * (getWidth() - 2 * padding.x) / 2d;
        Point c = getCenter();
        double phi = percentage * 2 * Math.PI;
        int x = (int) Math.round(r * Math.cos(phi - Math.PI / 2) + c.x);
        int y = (int) Math.round(r * Math.sin(phi - Math.PI / 2) + c.y);

        return new Point(x, y);
    }

    private Point getCrossPoint(double percentage) {
        return getCrossPoint(percentage, 1.0);
    }

    private Point getLabelPosition(double percentage) {
        return getCrossPoint(percentage, JDiagram.PERCENTAGE_RADIUS);
    }
}

import javax.swing.*;
import java.awt.*;
public class View extends JFrame {
    private JDiagram diagram;
    private JTextField[] fields = new JTextField[5];
    private Controller controller;
    public void initialise(Model model, Controller controller) {
        this.controller = controller;
        setTitle("Notenverteilung");

        addComponents();

        setVisible(true);

        diagram = new JDiagram();

            bottom.add(new JLabel((i + 1) + ":"));
            fields[i] = new JTextField("0", 5);

            bottom.add(fields[i]);
        }
        add(bottom, BorderLayout.SOUTH);
    }

@@solution Controller.java
import javax.swing.*;
import java.awt.event.*;
public class Controller extends KeyAdapter {
    private Model model;
    private View view;
    public void initialise(Model model, View view) { this.model = model; this.view = view; }
    @Override
    public void keyReleased(KeyEvent e) {
        if (e.getKeyCode() == KeyEvent.VK_ENTER) { return; }
        JTextField[] fields = view.getFields();
        int[] grades = new int[fields.length];
        try {
            for (int i = 0; i < fields.length; i++) {
                grades[i] = Integer.parseInt(fields[i].getText().trim());
            }
        } catch (NumberFormatException ex) {
            view.showError();
            return;
        }
        model.setGrades(grades);
    }
}

import javax.swing.*;
import java.awt.*;
import java.beans.*;
public class JDiagram extends JComponent implements PropertyChangeListener {
    private static final int BASE_PADDING = 10;
    private static final double PERCENTAGE_RADIUS = 0.5;
    private Model model;
    public void initialise(Model model) { this.model = model; model.addPropertyChangeListener(this); }
    @Override public void propertyChange(PropertyChangeEvent evt) { repaint(); }
    private Point getPadding() {
        int width = getWidth();
        int height = getHeight();

        if (width > height) {
            return new Point((width - height) / 2 + JDiagram.BASE_PADDING, JDiagram.BASE_PADDING);
        } else if (width < height) {
            return new Point(JDiagram.BASE_PADDING, (height - width) / 2 + JDiagram.BASE_PADDING);
        }

        return new Point(JDiagram.BASE_PADDING, JDiagram.BASE_PADDING);
    }

    private Point getCenter() {
        return new Point(getX() + getWidth()/2, getY() + getHeight()/2);
    }

    private Point getCrossPoint(double percentage, double percentageRadius) {
        Point padding = getPadding();
        double r = percentageRadius * (getWidth() - 2 * padding.x) / 2d;
        Point c = getCenter();
        double phi = percentage * 2 * Math.PI;
        int x = (int) Math.round(r * Math.cos(phi - Math.PI / 2) + c.x);
        int y = (int) Math.round(r * Math.sin(phi - Math.PI / 2) + c.y);

        return new Point(x, y);
    }

    private Point getCrossPoint(double percentage) {
        return getCrossPoint(percentage, 1.0);
    }

    private Point getLabelPosition(double percentage) {
        return getCrossPoint(percentage, JDiagram.PERCENTAGE_RADIUS);
    }
    @Override
    protected void paintComponent(Graphics g) {
        super.paintComponent(g);
        int[] grades = model.getGrades();
        int total = 0, nonZero = 0;
        for (int c : grades) { total += c; if (c > 0) nonZero++; }
        if (total == 0) return;
        Point p = getPadding();
        g.drawOval(p.x, p.y, getWidth() - 2 * p.x, getHeight() - 2 * p.y);
        double sum = 0;
        for (int i = 0; i < grades.length; i++) {
            if (grades[i] == 0) continue;
            double share = grades[i] / (double) total;
            if (nonZero > 1) {
                Point c = getCenter();
                Point q = getCrossPoint(sum);
                g.drawLine(c.x, c.y, q.x, q.y);
            }
            Point l = getLabelPosition(sum + share / 2);
            g.drawString(String.valueOf(i + 1), l.x, l.y);
            sum += share;
        }
    }
}
